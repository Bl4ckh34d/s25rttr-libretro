cmake_minimum_required(VERSION 3.24)
project(rttr_libretro LANGUAGES C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(PORT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../deps/s25rttr-android" CACHE PATH "Patched upstream Android source tree")
set(DEPS "${PORT_ROOT}/external")
set(GAME "${PORT_ROOT}/app/jni/s25client")
list(PREPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")

# Boost's modular trees provide targets; no host Boost or copied library bundle.
file(GLOB boost_modules LIST_DIRECTORIES TRUE "${DEPS}/boost/*")
set(RTTR_BOOST_INCLUDES "")
foreach(module IN LISTS boost_modules)
  if(EXISTS "${module}/include")
    list(APPEND RTTR_BOOST_INCLUDES "${module}/include")
  endif()
endforeach()
add_library(boost_headers INTERFACE)
target_include_directories(boost_headers INTERFACE ${RTTR_BOOST_INCLUDES})
add_library(Boost::boost ALIAS boost_headers)
add_library(Boost::headers ALIAS boost_headers)
add_library(boost_no_autolink INTERFACE)
add_library(Boost::disable_autolinking ALIAS boost_no_autolink)
include_directories(SYSTEM ${RTTR_BOOST_INCLUDES})
set(BOOST_LOCALE_ENABLE_ICU OFF CACHE BOOL "" FORCE)
foreach(module IN LISTS boost_modules)
  if(EXISTS "${module}/CMakeLists.txt")
    get_filename_component(name "${module}" NAME)
    add_subdirectory("${module}" "deps/boost/${name}" EXCLUDE_FROM_ALL)
  endif()
endforeach()

set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${DEPS}/SDL" deps/SDL EXCLUDE_FROM_ALL)
target_compile_definitions(SDL2-static PRIVATE RTTR_LIBRETRO)
set(SDL2MIXER_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(SDL2MIXER_INSTALL OFF CACHE BOOL "" FORCE)
set(SDL2MIXER_SAMPLES OFF CACHE BOOL "" FORCE)
set(SDL2MIXER_VENDORED ON CACHE BOOL "" FORCE)
set(SDL2MIXER_OPUS OFF CACHE BOOL "" FORCE)
set(SDL2MIXER_WAVPACK OFF CACHE BOOL "" FORCE)
set(SDL2MIXER_MOD OFF CACHE BOOL "" FORCE)
set(SDL2MIXER_MIDI_FLUIDSYNTH OFF CACHE BOOL "" FORCE)
add_subdirectory("${DEPS}/SDL_mixer" deps/SDL_mixer EXCLUDE_FROM_ALL)
target_compile_definitions(SDL2_mixer PRIVATE RTTR_LIBRETRO)

file(GLOB lua_sources "${DEPS}/lua/lua/*.c")
list(FILTER lua_sources EXCLUDE REGEX "/(lua|luac|onelua)\\.c$")
add_library(rttr_lua STATIC ${lua_sources})
target_include_directories(rttr_lua PUBLIC "${DEPS}/lua/lua")
set(LUA_INCLUDE_DIR "${DEPS}/lua/lua" CACHE PATH "" FORCE)
set(LUA_LIBRARY rttr_lua CACHE STRING "" FORCE)
set(LUA_LIBRARIES rttr_lua)

add_library(rttr_bz2 STATIC "${DEPS}/bzip2/blocksort.c" "${DEPS}/bzip2/huffman.c"
  "${DEPS}/bzip2/crctable.c" "${DEPS}/bzip2/randtable.c" "${DEPS}/bzip2/compress.c"
  "${DEPS}/bzip2/decompress.c" "${DEPS}/bzip2/bzlib.c")
set(BZ_VERSION "1.1.0")
configure_file("${DEPS}/bzip2/bz_version.h.in" "generated/bz_version.h" @ONLY)
target_include_directories(rttr_bz2 PUBLIC "${DEPS}/bzip2" PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
set(BZIP2_INCLUDE_DIR "${DEPS}/bzip2" CACHE PATH "" FORCE)
set(BZIP2_LIBRARY_RELEASE rttr_bz2 CACHE STRING "" FORCE)
add_library(BZip2::BZip2 ALIAS rttr_bz2)

set(UPNPC_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(UPNPC_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(UPNPC_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(UPNPC_BUILD_SAMPLE OFF CACHE BOOL "" FORCE)
add_subdirectory("${DEPS}/miniupnpc" deps/miniupnpc EXCLUDE_FROM_ALL)
file(COPY "${DEPS}/miniupnpc/include/" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/generated/miniupnpc")
set(MINIUPNPC_INCLUDE_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated" CACHE PATH "" FORCE)
set(MINIUPNPC_LIBRARY libminiupnpc-static CACHE STRING "" FORCE)

set(STATICLIB ON CACHE BOOL "" FORCE)
set(NOX11 ON CACHE BOOL "" FORCE)
set(NO_GBM ON CACHE BOOL "" FORCE)
set(NOEGL ON CACHE BOOL "" FORCE)
set(NO_LOADER ON CACHE BOOL "" FORCE)
set(NO_INIT_CONSTRUCTOR ON CACHE BOOL "" FORCE)
add_subdirectory("${DEPS}/gl4es" deps/gl4es EXCLUDE_FROM_ALL)
target_compile_definitions(GL PRIVATE GL4ES_COMPILE_FOR_USE_IN_SHARED_LIB)

set(RTTR_LIBRETRO ON CACHE BOOL "" FORCE)
set(RTTR_ENABLE_WERROR OFF CACHE BOOL "" FORCE)
set(RTTR_BUILD_UPDATER OFF CACHE BOOL "" FORCE)
set(RTTR_OPENGL GL4ES CACHE STRING "" FORCE)
set(RTTR_REVISION f5c16f1c127d3919f3c8f2336055c83f9184fb5a)
set(RTTR_BINDIR "." CACHE STRING "" FORCE)
set(RTTR_DATADIR "." CACHE STRING "" FORCE)
set(RTTR_GAMEDIR "S2" CACHE STRING "" FORCE)
set(RTTR_USE_STATIC_BOOST ON CACHE BOOL "" FORCE)
add_subdirectory("${GAME}" game EXCLUDE_FROM_ALL)
target_compile_definitions(s25Main PRIVATE RTTR_LIBRETRO)
target_include_directories(s25Main PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/include")

add_library(rttr_libretro SHARED core.cpp SaveStates.cpp VideoLibretro.cpp "${GAME}/extras/audioDrivers/SDL/AudioSDL.cpp")
target_include_directories(rttr_libretro PRIVATE include . "${GAME}/extras/audioDrivers/SDL" "${DEPS}/gl4es/include")
target_compile_definitions(rttr_libretro PRIVATE RTTR_LIBRETRO)
target_link_libraries(rttr_libretro PRIVATE s25Main videodrv audiodrv SDL2::SDL2-static SDL2_mixer::SDL2_mixer-static GL log GLESv2 EGL z)
target_link_options(rttr_libretro PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map" "-Wl,--exclude-libs,ALL")
set_target_properties(rttr_libretro PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
